\chapter{Polynomial Rings and Factorization}

Polynomials turn arithmetic in a ring into a tool for studying roots, maps,
and later linear operators.  Over a field, division of polynomials recovers
the Euclidean algorithm; this leads from fields to PIDs and UFDs.

\section{Polynomial Rings and Evaluation}
For a commutative ring \(R\), \(R[x]\) consists of finite sums
\(a_0+a_1x+\cdots+a_nx^n\), with coefficientwise addition and convolution
multiplication.  If \(R\) is a domain and \(f,g\ne0\), then
\(\deg(fg)=\deg f+\deg g\), because the product of leading coefficients is
nonzero.
\begin{proposition}[Degree Rule]
If \(R\) is a domain and \(f,g\in R[x]\) are nonzero, then
\(\deg(fg)=\deg f+\deg g\).  The units of \(R[x]\) are exactly the constant
polynomials whose value is a unit of \(R\).
\end{proposition}
\begin{proof}
Writing \(f=a_mx^m+\cdots\), \(g=b_nx^n+\cdots\), the coefficient of
\(x^{m+n}\) in \(fg\) is \(a_mb_n\), which is nonzero precisely because a
domain has no zero divisors.  Thus a product can have degree zero only when
both factors have degree zero, proving the assertion about units.
\end{proof}
\begin{proposition}[Evaluation] For \(a\in R\), evaluation
\(\operatorname{ev}_a:R[x]\to R\), \(f\mapsto f(a)\), is a ring
homomorphism.  If \(R\) is a field, its kernel is \((x-a)\). \end{proposition}
\begin{proof} The first assertion follows by distributing finite sums.  The
division algorithm below writes \(f=(x-a)q+r\), where \(r\) is constant;
then \(r=f(a)\), proving the kernel statement. \end{proof}

\section{Division Algorithms and Roots}
\begin{theorem}[Division Algorithm] If \(F\) is a field and \(f,g\in F[x]\),
\(g\ne0\), then there are unique \(q,r\in F[x]\) with
\[f=qg+r,\qquad r=0\ \text{or}\ \deg r<\deg g.\]
\end{theorem}
\begin{proof} Write the divisor and a current dividend as
\[
g=b_nx^n+b_{n-1}x^{n-1}+\cdots+b_0,
\qquad h=c_mx^m+c_{m-1}x^{m-1}+\cdots+c_0,
\]
where \(b_n\ne0\).  If \(m\ge n\), then
\[
(c_mb_n^{-1})x^{m-n}g
=c_mx^m+c_mb_n^{-1}b_{n-1}x^{m-1}+\cdots
+c_mb_n^{-1}b_0x^{m-n}.
\]
The first term is exactly the leading term of \(h\).  Subtracting this
polynomial therefore cancels the \(x^m\)-term and leaves a polynomial of
degree strictly smaller than \(m\).  Repeating this operation must stop,
since nonnegative degrees cannot decrease forever.  Collecting the multiples
of \(g\) that were subtracted gives \(f=qg+r\), with \(r=0\) or
\(\deg r<n\).

For uniqueness, suppose \(f=qg+r=q'g+r'\), where both remainders have
degree less than \(n\).  Then
\[
(q-q')g=r'-r.
\]
If \(q\ne q'\), the degree rule makes the left side have degree at least
\(\deg g=n\), whereas the right side has degree less than \(n\).  This is
impossible.  Hence \(q=q'\), and then \(r=r'\).
\end{proof}
\begin{example}[Long Division]
Dividing \(x^3-2x+4\) by \(x-1\), first subtract \(x^2(x-1)\), then
\(x(x-1)\), then \(-1(x-1)\).  Thus
\[x^3-2x+4=(x^2+x-1)(x-1)+3.\]
\end{example}
\begin{corollary}[Factor Theorem] For \(f\in F[x]\), \(f(a)=0\) iff
\(x-a\) divides \(f\). \end{corollary}
\begin{proof} Apply the division algorithm to \(f\) by \(x-a\) and evaluate.
\end{proof}
\begin{corollary} A nonzero polynomial of degree \(n\) over a field has at
most \(n\) roots. \end{corollary}
\begin{proof} Each root supplies a distinct linear factor; induction on
degree proves the assertion. \end{proof}

\section{Euclidean Domains and Principal Ideal Domains}
\begin{definition} A domain \(R\) is \emph{Euclidean} if it has a function
\(\delta:R\setminus\{0\}\to\mathbb N\) such that division with remainder
reduces \(\delta\).  A \emph{principal ideal domain} (PID) is a domain in
which every ideal is principal. \end{definition}
\begin{definition}
For \(a,b\) in a domain, a \emph{greatest common divisor} is an element \(d\)
such that \(d\mid a\), \(d\mid b\), and every common divisor of \(a,b\)
divides \(d\).  A gcd, when it exists, is unique only up to associates.  In
\(\mathbb Z\) one chooses the positive representative, but a general domain
need not have a preferred associate.
\end{definition}
\begin{theorem} Every Euclidean domain is a PID.  In particular \(F[x]\) is a
PID for every field \(F\). \end{theorem}
\begin{proof} Let \(I\ne0\) be an ideal and choose \(d\in I\) with minimal
\(\delta(d)\).  Divide any \(a\in I\) by \(d\): \(a=qd+r\).  Then
\(r\in I\), so minimality gives \(r=0\), hence \(I=(d)\).  Polynomial degree
is a Euclidean function by the division algorithm. \end{proof}
\begin{remark}[The implications are strict]
The implications
\[
\text{Euclidean domain}\Longrightarrow\text{PID}\Longrightarrow\text{UFD}
\]
cannot be reversed.  The second failure is exhibited below.  The first has
elementary but substantially longer examples; it is not needed for the
factorization theory developed here.  In
particular, ``not Euclidean'' means that no Euclidean function exists, not
merely that one familiar candidate fails.
\end{remark}
\begin{proposition}[B\'ezout and gcd]
In a PID, \((a,b)=(d)\) if and only if \(d\) is a gcd of \(a,b\), up to
associates.  Consequently every pair has a gcd and
\[d=ra+sb\]
for suitable \(r,s\).
\end{proposition}
\begin{proof}
If \((a,b)=(d)\), then \(a,b\in(d)\), so \(d\mid a,b\).  A common divisor
\(c\) divides every linear combination of \(a,b\), and hence divides
\(d\in(a,b)\).  Thus \(d\) is a gcd.  Conversely, write
\((a,b)=(g)\), which is possible because \(R\) is a PID.  The forward
direction shows that \(g\) is a gcd.  If \(d\) is another gcd, then
\(d\mid g\), since \(d\) is a common divisor, and \(g\mid d\), since \(g\)
is a common divisor.  Thus \(d\) and \(g\) are associates, so
\((d)=(g)=(a,b)\).  Consequently \(d\in(a,b)\), giving the displayed
B\'ezout representation.  This connects gcds, generated ideals, and
B\'ezout identities.
\end{proof}

\section{Unique Factorization Domains and Gauss's Lemma}
The factorization of an integer into primes is a model for a property enjoyed
by many domains.  The central question here is whether the property survives
the passage from a UFD \(R\) to \(R[x]\).  The answer is yes, and Gauss's
lemma explains why coefficients do not introduce hidden factorizations.

\begin{definition}
Let \(R\) be a domain.  Nonzero \(a,b\in R\) are \emph{associates} if
\(a=ub\) for a unit \(u\).  A nonzero nonunit \(p\) is \emph{irreducible} if
\(p=ab\) implies that \(a\) or \(b\) is a unit.  It is \emph{prime} if
\(p\mid ab\) implies \(p\mid a\) or \(p\mid b\).  A \emph{UFD} is a domain
where every nonzero nonunit has a factorization into irreducibles, unique up
to ordering and replacement by associates.
\end{definition}

Every prime is irreducible: if \(p=ab\), then \(p\mid a\) or \(p\mid b\);
cancellation makes the other factor a unit.  The converse need not hold in
an arbitrary domain, but it does hold in a PID.

\begin{proposition}[Irreducibles are prime in a PID]
Every irreducible element of a principal ideal domain is prime.
\end{proposition}
\begin{proof}
Let \(p\) be irreducible, suppose \(p\mid ab\), and assume \(p\nmid a\).
The gcd \(d\) of \(p,a\) cannot be associated to \(p\), so it is a unit.
Thus \((p,a)=R\), and B\'ezout gives \(rp+sa=1\).  After multiplication by
\(b\), both terms on the left of \(rpb+sab=b\) are divisible by \(p\).
Hence \(p\mid b\).
\end{proof}

\phantomsection\label{thm:irreducible-maximal-field}
\begin{applicationtheorem}[Irreducible polynomials and field quotients]
Let \(F\) be a field and let \(f\in F[x]\) be nonconstant.  The following are
equivalent:
\[
f\text{ is irreducible},
\qquad
(f)\text{ is a maximal ideal of }F[x],
\qquad
F[x]/(f)\text{ is a field}.
\]
\end{applicationtheorem}
\begin{proof}
Since \(F[x]\) is a PID, every ideal containing \((f)\) has the form
\((g)\), where \(g\mid f\).  If \(f\) is irreducible, then \(g\) is either a
unit or an associate of \(f\).  Thus there is no ideal strictly between
\((f)\) and \(F[x]\), so \((f)\) is maximal.  Conversely, if
\(f=gh\) with both \(g\) and \(h\) nonunits, then
\[
(f)\subsetneq(g)\subsetneq F[x],
\]
contradicting maximality.  The equivalence between maximal ideals and field
quotients is the quotient characterization of maximal ideals from Chapter~3.
\end{proof}

Irreducibility therefore does more than forbid a factorization: it is exactly
what makes the quotient obtained by adjoining a formal root into a field.

\begin{theorem}[Every PID is a UFD]
Every principal ideal domain is a unique factorization domain.
\end{theorem}
\begin{proofidea}
Existence comes from the ascending-chain condition on ideals: an element that
could not be factored would force a strictly increasing chain of principal
ideals.  Uniqueness follows by cancelling prime factors.
\end{proofidea}
\begin{proof}
An ascending chain of ideals in a PID stabilizes, since its union is an ideal
\((d)\), and \(d\) belongs to one member of the chain.  Suppose some nonzero
nonunit does not factor into irreducibles.  If the collection of principal
ideals generated by such elements had no maximal member, beginning with any
one of them and repeatedly choosing a strictly larger member would give a
nonstabilizing ascending chain.  Thus choose \(a\) with \((a)\) maximal
among these ideals.  Then \(a=bc\), with both
\(b,c\) nonunits.  Therefore \((a)\subsetneq(b)\) and
\((a)\subsetneq(c)\), so maximality gives factorizations of \(b,c\), a
contradiction.  Thus factorizations exist.

If \(p_1\cdots p_m=q_1\cdots q_n\) are products of irreducibles, \(p_1\) is
prime by the proposition, hence divides some \(q_j\).  Since \(q_j\) is
irreducible, they are associates.  Reorder, absorb the unit, and cancel.
Induction gives uniqueness.
\end{proof}

\begin{example}[A UFD that is not a PID]
For a field \(k\), the polynomial UFD theorem below will show that \(k[x,y]\)
is a UFD; this is a forward preview.  It is not a PID: if \((x,y)=(f)\), then \(f\mid x\) and
\(f\mid y\).  Since \(x\) and \(y\) are irreducible and not associates,
\(f\) must be a unit, which would make \((x,y)=k[x,y]\).  But every element
of \((x,y)\) has zero constant term, so \(1\notin(x,y)\).
\end{example}

Let \(R\) now be a UFD.  A gcd of coefficients is only determined up to a
unit, precisely the ambiguity that factorization theory permits.

Gcds exist in every UFD, although they need not have B\'ezout
representations.  After choosing representatives of irreducible associate
classes, write
\[a=u\prod p_i^{\alpha_i},\qquad b=v\prod p_i^{\beta_i}.\]
Then, up to associates,
\[\gcd(a,b)\sim\prod p_i^{\min(\alpha_i,\beta_i)}.\]
The exponents show at once that this element divides both \(a\) and \(b\),
and that every common divisor divides it.  In a PID, gcds are B\'ezout
combinations; in a general UFD this is false.  For example,
\(\gcd(x,y)=1\) in \(F[x,y]\), but \((x,y)\ne F[x,y]\): an identity
\(fx+gy=1\) would become \(0=1\) after evaluation at \(x=y=0\).
Thus
\[
\gcd(a,b)=1\quad\not\Rightarrow\quad(a,b)=R
\]
in a general UFD.  In a PID, B\'ezout's identity gives the equivalence
\[\gcd(a,b)=1\quad\Longleftrightarrow\quad(a,b)=R.\]
Equivalently, the principal ideals \((a)\) and \((b)\) are comaximal exactly
when \((a,b)=R\).  Thus \(k[x,y]\) is a UFD but not a PID for the same reason
that its coprime elements \(x,y\) admit no B\'ezout identity.

\begin{definition}[Content]
For \(0\ne f=a_0+\cdots+a_nx^n\in R[x]\), its \emph{content} is
\[\operatorname{cont}(f)=\gcd(a_0,\ldots,a_n),\]
where the gcd is understood up to multiplication by a unit.  The polynomial
is \emph{primitive} exactly when \(\operatorname{cont}(f)\) is a unit,
equivalently when no nonunit divides all of its coefficients.
\end{definition}

The content is therefore simply the gcd of the coefficients, whose existence
is supplied by the UFD property just discussed.

Every nonzero \(f\) has the form
\[f=\operatorname{cont}(f)f_0\]
up to multiplication by a unit, with \(f_0\) primitive.
For example,
\[6x^2-9x+3=3(2x^2-3x+1),\]
and the parenthesized factor is primitive in \(\mathbb Z[x]\).

\begin{lemma}[Primitive Product Lemma]
The product of two primitive polynomials in \(R[x]\) is primitive.
\end{lemma}
\begin{proof}
If a nonunit divided every coefficient of \(fg\), some irreducible factor
\(p\) would do so.  In a UFD, irreducibles are prime, so \((p)\) is a prime
ideal: \(ab\in(p)\) means \(p\mid ab\).  Hence \(R/(p)\) is a domain by
Chapter~3.  The reductions of \(f\) and \(g\) modulo \(p\) are nonzero,
because \(f,g\) are primitive, but their product is zero.  This contradicts
the degree rule in the polynomial ring over a domain.
\end{proof}

\begin{corollary}[Content formula]
For nonzero \(f,g\in R[x]\),
\[\operatorname{cont}(fg)\sim\operatorname{cont}(f)\operatorname{cont}(g),\]
where \(\sim\) denotes association.  Equality is naturally only up to
associates in a general UFD.  For \(R=\mathbb Z\), normalizing content to
be positive gives the literal equality
\(\operatorname{cont}(fg)=\operatorname{cont}(f)\operatorname{cont}(g)\).
\end{corollary}
\begin{proof}
Write \(f=\operatorname{cont}(f)f_0\),
\(g=\operatorname{cont}(g)g_0\), where \(f_0,g_0\) are primitive.
Then \(fg=\operatorname{cont}(f)\operatorname{cont}(g)f_0g_0\), and the
lemma says the final factor has unit
content.
\end{proof}

Let \(K=\operatorname{Frac}(R)\).  The next statement is the precise form
of clearing denominators.

\begin{lemma}[Gauss's factorization lemma]
If \(f\in R[x]\) is primitive and \(f=gh\) in \(K[x]\), then there are
primitive \(g_0,h_0\in R[x]\) and a unit \(\varepsilon\in R\) such that
\[f=(\varepsilon g_0)h_0.\]
Thus a positive-degree factorization over \(K\) yields one over \(R\).
\end{lemma}
\begin{proof}
Choose nonzero \(a,b\in R\) such that \(G=ag\) and \(H=bh\) lie in \(R[x]\).
Write \(G=\operatorname{cont}(G)g_0\),
\(H=\operatorname{cont}(H)h_0\) with primitive parts.  Since
\(GH=abf\), the content formula gives
\[\operatorname{cont}(G)\operatorname{cont}(H)
\sim\operatorname{cont}(GH)=\operatorname{cont}(abf)\sim ab.\]
Thus \(\operatorname{cont}(G)\operatorname{cont}(H)=\varepsilon ab\) for a
unit \(\varepsilon\), and
\[abf=GH=\varepsilon abg_0h_0.\]
Cancellation in the domain \(R[x]\) gives the result.  Scaling does not
change degree.
\end{proof}

\begin{corollary}[Irreducibility transfer]
For primitive \(f\in R[x]\),
\[f\text{ is irreducible in }R[x]\quad\Longleftrightarrow\quad
f\text{ is irreducible in }K[x].\]
\end{corollary}
\begin{proof}
A factorization in \(R[x]\) is one in \(K[x]\).  Conversely, a nontrivial
factorization in \(K[x]\) has two positive-degree factors; the lemma turns it
into one in \(R[x]\).
\end{proof}

\begin{lemma}
If primitive \(g,h\in R[x]\) are associates in \(K[x]\), then they are
associates in \(R[x]\).
\end{lemma}
\begin{proof}
Write \(h=(a/b)g\) with \(a,b\in R\setminus\{0\}\).  Then \(bh=ag\).
Taking contents gives \(b\sim a\), so \(a/b\) is a unit of \(R\).
\end{proof}

\begin{theorem}[Polynomial UFD Theorem]
For a domain \(R\),
\[
R\text{ is a UFD}\quad\Longleftrightarrow\quad R[x]\text{ is a UFD}.
\]
Consequently, for every finite \(n\ge1\),
\[
R\text{ is a UFD}\quad\Longleftrightarrow\quad
R[x_1,\ldots,x_n]\text{ is a UFD}.
\]
\end{theorem}
\begin{proof}
The field \(K\) has \(K[x]\) a PID, hence a UFD.  For \(0\ne f\in R[x]\),
factor its content in \(R\), then factor its primitive part in \(K[x]\).
The factorization lemma clears the denominators of the latter factors; the
irreducibility-transfer corollary makes the resulting primitive factors
irreducible in \(R[x]\).  Irreducible constants of \(R\) remain irreducible
in \(R[x]\), by the degree rule.  This proves existence.

For uniqueness, compare two factorizations in the UFD \(K[x]\).  Its
uniqueness pairs the primitive factors up to association, and the preceding
lemma upgrades these associations to \(R[x]\).  Unique factorization of the
remaining contents occurs in \(R\).  Repeated application of the one-variable
result gives the forward multivariable assertion.

Conversely, suppose \(R[x]\) is a UFD.  A factorization of a nonzero constant
polynomial cannot involve a positive-degree factor, by the degree rule.  A
constant polynomial is a unit in \(R[x]\) precisely when its value is a unit
in \(R\).  Existence and uniqueness of factorizations of constants in \(R[x]\)
therefore give existence and uniqueness in \(R\).  The multivariable converse
follows by viewing constants successively in each one-variable polynomial ring.
\end{proof}

\begin{corollary}
\(\mathbb Z[x]\) and \(\mathbb Q[x]\) are UFDs.  A primitive polynomial in
\(\mathbb Z[x]\) is irreducible over \(\mathbb Z\) exactly when it is
irreducible over \(\mathbb Q\).
\end{corollary}
\begin{proof}
Apply the theorem to the UFD \(\mathbb Z\), whose fraction field is
\(\mathbb Q\), and use irreducibility transfer.
\end{proof}

\section{Irreducibility Criteria and Factorization over Fields}
\begin{proposition}
A polynomial of degree two or three over a field is reducible if and only if
it has a root in that field.
\end{proposition}
\begin{proof} A root yields a linear factor by the Factor Theorem.  Conversely,
a proper factorization of degree two or three has a factor of degree one,
which has a root. \end{proof}
\begin{theorem}[Eisenstein's Criterion]
Let \(f=a_nx^n+\cdots+a_0\in\mathbb Z[x]\).  If a prime \(p\) divides every
\(a_i\) for \(i<n\), does not divide \(a_n\), and \(p^2\nmid a_0\), then
\(f\) is irreducible over \(\mathbb Q\).
\end{theorem}
\begin{proof} By Gauss's Lemma it is enough to rule out a factorization in
\(\mathbb Z[x]\); the factors may be chosen primitive.  Suppose that
\(f=gh\) is such a factorization with both factors of positive degree.  Since
\(p\nmid a_n\), and the leading coefficients of \(g\) and \(h\) multiply to
\(a_n\), neither leading coefficient is divisible by \(p\).  Reduction modulo
\(p\) therefore preserves the positive degrees of both factors.  It gives
\[\overline g\,\overline h=\overline {a_n}x^n.\]
Since \(\mathbb F_p[x]\) is a domain, comparison of the smallest powers of
\(x\) occurring shows
\(\overline g=cx^r\), \(\overline h=dx^s\), with \(c,d\ne0\) and
\(r,s>0\).  Thus the constant terms of both \(g\) and \(h\) are divisible by
\(p\), so \(p^2\) divides their product \(a_0\), a contradiction.
\end{proof}
\begin{example} Eisenstein at \(p=2\) proves \(x^n-2\) irreducible over
\(\mathbb Q\) for every \(n\ge1\). \end{example}

\section{Polynomials over Finite Fields}
Over \(\mathbb F_p\), and more generally over a finite field \(\mathbb F_q\),
the preceding irreducibility tests and division algorithm work without change.
Factorization depends on the coefficient field: \(x^2+1\) is irreducible in
\(\mathbb F_3[x]\), since it has no root there, but
\(x^2+1=(x-2)(x+2)\) in \(\mathbb F_5[x]\).

If \(f\in\mathbb F_q[x]\) is irreducible of degree \(n\), then
\[E=\mathbb F_q[x]/(f)\]
is a field.  Division by \(f\) gives a unique representative of degree less
than \(n\), so there are exactly \(q^n\) elements in \(E\).  This is the
concrete construction of a field customarily denoted \(\mathbb F_{q^n}\).
For every \(z\in E\), Lagrange's theorem applied to \(E^\times\) gives
\(z^{q^n-1}=1\) when \(z\ne0\); hence the roots of
\[
x^{q^n}-x
\]
are precisely the elements of \(E\).  They are distinct because its
derivative is \(-1\).  The systematic uniqueness and subfield theory of
finite fields will be proved in Chapter~10.  As a preview of that theory,
over \(\mathbb F_q\) one has
\[
x^{q^n}-x=\prod \{h(x):h\text{ monic irreducible and }\deg h\mid n\}.
\]
The product is over distinct polynomials.  Indeed, a root of an irreducible
polynomial of degree \(d\) lies in \(\mathbb F_{q^n}\) exactly when \(d\mid n\);
the promised finite-field theory supplies this final equivalence.

\section{A Short Quadratic-Residue Recap}
Let \(p\) be an odd prime.  A nonzero class \(a\in\mathbb F_p\) is a
\emph{quadratic residue} if \(a=b^2\) for some \(b\ne0\).  Exactly
\((p-1)/2\) nonzero classes are squares: the map \(b\mapsto b^2\) on
\(\mathbb F_p^\times\) has fibers \(\{b,-b\}\).  The Legendre symbol is
\[
\left(\frac ap\right)=
\begin{cases}1,&a\text{ is a nonzero square modulo }p,\\-1,&a\text{ is a nonsquare modulo }p,\\0,&p\mid a.\end{cases}
\]

\begin{theorem}[Euler's criterion]
For \(p\nmid a\),
\[
a^{(p-1)/2}\equiv\left(\frac ap\right)\pmod p.
\]
\end{theorem}
\begin{proof}
Put \(m=(p-1)/2\).  The \(m\) nonzero squares are roots of \(x^m-1\), so
they are exactly its roots.  Since \(x^{2m}-1=(x^m-1)(x^m+1)\) and every
nonzero element satisfies \(x^{2m}=1\), the nonsquares are exactly the roots
of \(x^m+1\).  This proves the assertion.  Equivalently, it describes the
index-two subgroup of \(\mathbb F_p^\times\); its cyclicity was proved in
the group-theory chapter.
\end{proof}
For \(p\nmid ab\), Euler's criterion also gives the basic multiplicativity
in one line:
\[
\left(\frac{ab}{p}\right)
\equiv(ab)^{(p-1)/2}
\equiv a^{(p-1)/2}b^{(p-1)/2}
\equiv\left(\frac ap\right)\left(\frac bp\right)\pmod p.
\]
Both sides are \(\pm1\), so they are equal.
The supplementary laws are
\[
\left(\frac{-1}{p}\right)=(-1)^{(p-1)/2},\qquad
\left(\frac2p\right)=(-1)^{(p^2-1)/8}.
\]
The first follows at once from Euler's criterion; the second is recorded for
useful orientation.  For distinct odd primes \(p,q\), quadratic reciprocity
states
\[
\left(\frac pq\right)\left(\frac qp\right)
=(-1)^{(p-1)(q-1)/4}.
\]
Thus the symbols agree unless \(p\equiv q\equiv3\pmod4\), in which case they
are opposite.  For example, \((3/5)=-1=(5/3)\), while
\((3/7)=-1\) and \((7/3)=1\).

\begin{remark}[Two viewpoints on quadratic reciprocity]
We do not prove quadratic reciprocity here.  In Gauss's classical lemma, one
studies the least residues of \(a,2a,\ldots,((p-1)/2)a\) and counts those
falling in the negative half of a residue system; the count's parity gives
the Legendre symbol.  Gauss's lattice-point version counts points in a
rectangle separated by a line of slope \(q/p\), and counting from the two
directions gives the reciprocity sign.  This is an arithmetic/counting
viewpoint.  There is also a genuinely algebraic one: factorization identities
in characteristic \(p\), the Frobenius map \(z\mapsto z^p\), and the
quadratic-residue information above combine in polynomial rings over finite
fields to yield the same law.  Dummit--Foote develops this polynomial proof;
we omit it because reciprocity is an application, not a prerequisite.  Many
other proofs are known, but Gauss sums, cyclotomic fields, and algebraic
number theory lie beyond this chapter.
\end{remark}

\section{Chain Conditions and Noetherian Rings}
Ideals can be complicated, so a basic finiteness question is whether they can
be controlled by finitely much data.  The next conditions rule out endlessly
increasing or decreasing sequences of genuinely new ideals.  A ring
satisfies the \emph{ascending chain condition} (ACC) on ideals if every chain
\[
I_1\subseteq I_2\subseteq I_3\subseteq\cdots
\]
stabilizes: there is an \(n\) such that \(I_k=I_{k+1}\) for every \(k\ge n\),
or, equivalently, \(I_n=I_{n+1}=I_{n+2}=\cdots\).  It satisfies the
\emph{descending chain condition} (DCC) if every chain
\[
I_1\supseteq I_2\supseteq I_3\supseteq\cdots
\]
stabilizes in the analogous sense.  A commutative ring is \emph{Noetherian}
if it satisfies ACC on ideals and \emph{Artinian} if it satisfies DCC on
ideals.

The characterization below shows that Noetherianity is precisely the
principle that every ideal is finitely generated.  It is therefore more than
an abstract chain condition: it supplies finite control in arguments about
ideals.  Noetherian rings are central in later commutative algebra, where
finite generation, maximal-element arguments, and terminating ascending
constructions recur; polynomial rings, quotient rings, and localizations
provide major sources.  Artinian rings impose finiteness in the opposite
direction.  They too are important, especially in finite-length and local
situations, but their deeper structure theory belongs to a later course.

Every field is both Noetherian and Artinian, since it has only the ideals
\((0)\) and \((1)\).  Every finite ring is both, because a strictly changing
chain would contain infinitely many distinct ideals.  The ring \(\mathbb Z\)
is Noetherian but not Artinian: its ideals are principal, whereas
\[
(2)\supsetneq(4)\supsetneq(8)\supsetneq\cdots.
\]
Likewise \(F[x]\) is not Artinian, because
\((x)\supsetneq(x^2)\supsetneq(x^3)\supsetneq\cdots\).  We do not pursue
the deeper structure theory of Artinian rings here.

\begin{theorem}[Characterizations of Noetherian rings]
For a commutative ring \(R\), the following are equivalent:
\begin{enumerate}
\item \(R\) satisfies ACC on ideals;
\item every nonempty collection of ideals has a maximal member under inclusion;
\item every ideal of \(R\) is finitely generated.
\end{enumerate}
\end{theorem}
\begin{proof}
If (i) holds and a nonempty collection has no maximal member, choose an ideal
in it and then a strictly larger one repeatedly; this produces a nonstabilizing
ascending chain, a contradiction.  Thus (i) implies (ii).

Assume (ii), and let \(I\) be an ideal.  Among the finitely generated ideals
contained in \(I\), choose a maximal one, say \(J=(a_1,\ldots,a_r)\).  If
\(J\ne I\), choose \(a\in I\setminus J\).  Then
\((a_1,\ldots,a_r,a)\) is a strictly larger finitely generated ideal in
\(I\), a contradiction.  Hence \(I=J\), proving (iii).

Finally assume (iii), and let \(I_1\subseteq I_2\subseteq\cdots\).  Their
union \(I\) is an ideal, hence has generators \(a_1,\ldots,a_r\).  All of
these generators belong to some common \(I_N\); then \(I_N=I\), and every
subsequent ideal is also \(I\).  Thus the chain stabilizes.
\end{proof}

The three formulations support different proof styles.  ACC is useful when
an argument builds larger and larger ideals, the maximal-member formulation
is useful for choosing a maximal counterexample, and finite generation is the
most concrete algebraic form.  This pattern has already appeared implicitly
in the proof that a PID is a UFD.

\section{Hilbert's Basis Theorem}
Polynomial rings are among the most fundamental constructions in algebra, so
it matters that adjoining finitely many variables preserves this finiteness
principle.  Hilbert's Basis Theorem is thus much more than another source of
Noetherian rings: by iteration it makes every \(R[x_1,\ldots,x_n]\)
Noetherian when \(R\) is.  In particular, \(F[x_1,\ldots,x_n]\) is
Noetherian for every field \(F\).  The theorem is foundational in later
commutative algebra and algebraic geometry, where polynomial rings and their
quotients are basic objects.
\begin{theorem}[Hilbert's Basis Theorem]
If \(R\) is Noetherian, then \(R[x]\) is Noetherian.
\end{theorem}
\begin{proof}
The zero ideal is finitely generated, so let \(I\ne0\) be an ideal of
\(R[x]\).  Choose \(f_1\in I\) of minimal
degree.  Having chosen \(f_1,\ldots,f_i\), if
\((f_1,\ldots,f_i)\ne I\), choose
\[
f_{i+1}\in I\setminus(f_1,\ldots,f_i)
\]
of minimal degree.  The degrees are nondecreasing.  Indeed,
\[
I\setminus(f_1,\ldots,f_i)
\subseteq I\setminus(f_1,\ldots,f_{i-1}),
\]
and \(f_i\) was chosen with minimum degree in the larger set.

Write \(a_i=\operatorname{LC}(f_i)\).  The ideals
\[
(a_1)\subseteq(a_1,a_2)\subseteq(a_1,a_2,a_3)\subseteq\cdots
\]
stabilize because \(R\) is Noetherian.  Suppose that they stabilize from
\(n\) onward but that \((f_1,\ldots,f_n)\ne I\).  The construction then
gives \(f_{n+1}\notin(f_1,\ldots,f_n)\), while
\[
a_{n+1}=u_1a_1+\cdots+u_na_n
\]
for suitable \(u_1,\ldots,u_n\in R\).  Since
\(\deg f_j\le\deg f_{n+1}\) for \(j\le n\), the polynomial
\[
g=f_{n+1}-\sum_{j=1}^n
u_jx^{\deg f_{n+1}-\deg f_j}f_j
\]
belongs to \(I\) and has degree strictly smaller than \(\deg f_{n+1}\),
unless it is zero.  If \(g\in(f_1,\ldots,f_n)\), then the displayed equation
would put \(f_{n+1}\) in that ideal, a contradiction.  Hence
\(g\in I\setminus(f_1,\ldots,f_n)\), and its smaller degree contradicts the
minimal choice of \(f_{n+1}\).  Thus the construction terminates, so every
ideal of \(R[x]\) is finitely generated.
\end{proof}
The proof illustrates a recurring strategy: from complicated data in
\(R[x]\), extract coefficient data in \(R\), use Noetherianity there, and
lift the resulting finite control back to \(R[x]\).  Here the recursive
minimal-degree choice turns stabilization of leading-coefficient ideals into
a leading-term cancellation, whose smaller degree gives the contradiction.
\begin{corollary}
If \(R\) is Noetherian, then \(R[x_1,\ldots,x_n]\) is Noetherian for every
finite \(n\).  In particular, \(F[x_1,\ldots,x_n]\) is Noetherian for every
field \(F\).
\end{corollary}
\begin{proof}
Apply the theorem first to \(R\), then to \(R[x]\), and continue.
\end{proof}

Thus, for \(n\ge2\), \(F[x_1,\ldots,x_n]\) is a Noetherian UFD but not a
PID.  Hilbert's theorem guarantees that its ideals have finite generating
sets.  This is an existence theorem; Gr\"obner-basis theory, introduced
below, asks for generators adapted to a monomial order so that ideal
membership and related computations become effective.

\section{Formal Power Series}
Polynomial rings allow only finitely many nonzero coefficients.  Formal power
series allow infinitely many while remaining entirely algebraic: there is no
analytic convergence, equality and operations are coefficientwise, and each
coefficient of a product uses only finitely many coefficient products.  They
arise naturally later in local algebra, completions, and the study of local
behavior.
For a commutative ring \(R\), the ring \(R[[x]]\) consists of formal
expressions
\[f=\sum_{n=0}^\infty a_nx^n,\qquad a_n\in R.\]
There is no convergence assertion: equality is defined coefficientwise,
\[
\sum_{n=0}^\infty a_nx^n=\sum_{n=0}^\infty b_nx^n
\quad\Longleftrightarrow\quad a_n=b_n\text{ for every }n.
\]
Addition is coefficientwise and multiplication is the Cauchy product
\[
\left(\sum_{n=0}^\infty a_nx^n\right)
\left(\sum_{n=0}^\infty b_nx^n\right)
=\sum_{n=0}^\infty\left(\sum_{i=0}^n a_ib_{n-i}\right)x^n.
\]
The coefficient of \(x^n\) uses only finitely many products, so this is an
algebraic definition.  For \(n\ge2\), define finitely many variables
inductively by
\[
R[[x_1,\ldots,x_n]]\coloneqq R[[x_1,\ldots,x_{n-1}]][[x_n]].
\]
Thus \(R[[x,y]]=(R[[x]])[[y]]\).  This agrees naturally with the direct
multi-index description
\[
R[[x_1,\ldots,x_n]]=
\left\{\sum_{\alpha\in\mathbb N^n}a_\alpha x^\alpha:a_\alpha\in R\right\},
\qquad x^\alpha=x_1^{\alpha_1}\cdots x_n^{\alpha_n}.
\]
For example, the correspondence for two variables is
\[
\sum_{j\ge0}\left(\sum_{i\ge0}a_{ij}x^i\right)y^j
\quad\longleftrightarrow\quad
\sum_{i,j\ge0}a_{ij}x^iy^j.
\]
Conversely, one groups a double family by the exponent of \(y\).  This
identification respects addition and multiplication.  Indeed, for fixed
\(\gamma\in\mathbb N^n\), there are only
finitely many pairs \(\alpha,\beta\in\mathbb N^n\) with
\(\alpha+\beta=\gamma\), so the coefficient of \(x^\gamma\) in a product is
\[\sum_{\alpha+\beta=\gamma}a_\alpha b_\beta.\]
For example,
\[(1-x)^{-1}=1+x+x^2+x^3+\cdots\]
in \(R[[x]]\), since multiplication by \(1-x\) has constant coefficient
one and every higher coefficient zero.

\begin{proposition}[Units in a power-series ring]
The series \(f=a_0+a_1x+\cdots\in R[[x]]\) is a unit if and only if \(a_0\)
is a unit in \(R\).
\end{proposition}
\begin{proof}
If \(fg=1\), comparison of constant coefficients gives
\(a_0b_0=1\).  Conversely, if \(a_0\) is a unit, construct
\(g=b_0+b_1x+\cdots\) recursively: take \(b_0=a_0^{-1}\), and, after
choosing \(b_0,\ldots,b_{n-1}\), set
\[
b_n=-a_0^{-1}\sum_{i=1}^n a_i b_{n-i}.
\]
Then the coefficient of \(x^n\) in \(fg\) is zero for \(n>0\), while its
constant coefficient is one.  Hence \(fg=1\) term by term.
\end{proof}
\begin{corollary}
For a field \(F\), the ring \(F[[x]]\) is local with unique maximal ideal
\((x)\).
\end{corollary}
\begin{proof}
A series is a unit exactly when its constant coefficient is nonzero.  Thus
the nonunits are exactly the series with zero constant coefficient, namely
\((x)\); use the characterization of local rings from Chapter~3.
\end{proof}

Hilbert's theorem shows that Noetherianity survives polynomial extension;
formal power series are another fundamental enlargement of a ring, so it is
natural to ask whether the same finite-control property survives here.
\begin{theorem}
If \(R\) is Noetherian, then \(R[[x]]\) is Noetherian.
\end{theorem}
\begin{proof}
Let \(I\) be an ideal of \(R[[x]]\).  For \(n\ge0\), define
\[
I_n=\{a\in R:\text{ some }f\in I\text{ has the form }
f=ax^n+\text{terms of degree }>n\}.
\]
The zero series shows that \(0\in I_n\).  Adding two witnessing series, or
multiplying one by an element of \(R\), shows that \(I_n\) is an ideal of
\(R\).  Multiplication of a witnessing series by \(x\) gives
\(I_n\subseteq I_{n+1}\).  Hence this chain stabilizes: choose \(m\) with
\(I_m=I_{m+1}=\cdots\).

For each \(0\le n\le m\), choose generators
\(a_{n1},\ldots,a_{nr_n}\) of \(I_n\), and choose witnessing series
\(f_{ni}\in I\) beginning with \(a_{ni}x^n\).  Let \(J\) be the ideal of
\(R[[x]]\) generated by this finite collection.  We prove \(I\subseteq J\).
Given \(f\in I\), cancel its coefficient of \(x^0\) using the \(f_{0i}\).
At stage \(j\), the remaining series has zero coefficients below \(j\), and
its coefficient \(c_j\) belongs to \(I_j\).  If \(j\le m\), express \(c_j\)
using the \(a_{ji}\); if \(j>m\), use the generators of \(I_m=I_j\) and
multiply their witnessing series by \(x^{j-m}\).  In either case, subtract a
finite combination of the selected \(f_{ni}\) whose degree-\(j\) term is
\(c_jx^j\).  The resulting series has zero coefficients through degree
\(j\).

This recursive cancellation may be infinite, but it is formal rather than
topological.  For each fixed degree, only the finitely many corrections from
the preceding stages can affect its coefficient.  Grouping the correction
terms by the finite set of generators therefore produces series
\(h_{ni}\in R[[x]]\) with
\[
f=\sum_{n=0}^m\sum_{i=1}^{r_n}h_{ni}f_{ni}.
\]
Indeed, coefficientwise this is merely an interchange of an infinite sum with
a finite sum, and every coefficient receives finitely many contributions.
The difference between the two sides has every coefficient zero, hence is
zero by the definition of equality in \(R[[x]]\).  Thus \(f\in J\).  The
reverse containment is clear because the \(f_{ni}\) lie in \(I\), so
\(I=J\) is finitely generated.
\end{proof}
\begin{corollary}
If \(R\) is Noetherian, then
\[
R[[x_1,\ldots,x_n]]
\]
is Noetherian for every finite \(n\).
\end{corollary}
\begin{proof}
Apply the one-variable theorem successively along
\[
R\longmapsto R[[x_1]]\longmapsto R[[x_1,x_2]]\longmapsto\cdots
\longmapsto R[[x_1,\ldots,x_n]].
\]
\end{proof}

The two Noetherianity proofs share the same coefficient strategy but use the
opposite ends of a series.  A polynomial has a highest degree, so one controls
leading coefficients and cancellation decreases degree.  A formal power
series generally has no highest degree; one instead controls lowest-order
coefficients and cancels in increasing order.  The latter process may be
infinite, but it is legitimate coefficientwise: every fixed coefficient has
only finitely many relevant contributions.  In both cases, coefficient data
in \(R\) is what transfers Noetherianity to the larger ring.

This is only a first encounter with Noetherian methods.  In later
commutative algebra, Noetherian rings provide a central setting for studying
ideals, modules, localization, dimension, local rings, completions, and
related structural questions.  Hilbert's theorem and the formal-power-series
theorem are early instances of the recurring principle that useful finiteness
properties should survive natural algebraic constructions.

\section{Polynomial Rings in Several Variables}
For a commutative ring \(R\), the ring \(R[x_1,\ldots,x_n]\) consists of
finite sums \(\sum_\alpha a_\alpha x^\alpha\), where
\[
x^\alpha=x_1^{\alpha_1}\cdots x_n^{\alpha_n},\qquad
\alpha=(\alpha_1,\ldots,\alpha_n)\in\mathbb N^n.
\]
Addition is coefficientwise and multiplication uses \(x^\alpha x^\beta=
x^{\alpha+\beta}\).  The \emph{total degree} of \(x^\alpha\) is
\(|\alpha|=\alpha_1+\cdots+\alpha_n\), and the total degree of a polynomial
is the largest total degree of one of its nonzero terms.  For a field \(F\)
and \(n\ge2\), repeated Gauss's theorem makes \(F[x_1,\ldots,x_n]\) a UFD,
but it is neither a PID nor a Euclidean domain.  Indeed,
\((x_1,x_2)\) is not principal by the same argument used above for \((x,y)\).
Since every Euclidean domain is a PID, it cannot be Euclidean.

\section{An Introduction to Gr\"obner Bases}
For the remainder of this section, the coefficient ring is a field \(F\), and
we work in \(F[x_1,\ldots,x_n]\).  In more than one variable, division needs
an agreed notion of the largest monomial.  A \emph{monomial order} is a total
order on the monomials that is a well-order, is compatible with
multiplication, and has \(1\) as its least element.  Lexicographic order compares the
first exponent at which two exponent vectors differ; graded lexicographic
order compares total degree first, then breaks ties lexicographically.  Once
an order is fixed, \(\operatorname{LC}(f)\), \(\operatorname{LM}(f)\), and
\(\operatorname{LT}(f)\) denote respectively the leading coefficient,
monomial, and term of \(f\).

Given an ordered list \(g_1,\ldots,g_s\), multivariable division repeatedly
cancels the leading term of the current dividend when some
\(\operatorname{LM}(g_i)\) divides it; otherwise that leading term is moved
to the remainder.  Unlike one-variable division, the remainder can depend on
the chosen divisors and their order.  For an ideal \(I\), its
\emph{leading-term ideal} is
\[\operatorname{LT}(I)=(\operatorname{LT}(f):0\ne f\in I).\]
A finite set \(G\subseteq I\) is a \emph{Gr\"obner basis} if
\(\operatorname{LT}(I)=(\operatorname{LT}(g):g\in G)\).  Then division by
\(G\) gives the useful membership test
\[f\in I\quad\Longleftrightarrow\quad\text{the remainder on division by }G
\text{ is }0.\]

For nonzero \(f,g\), let \(M=\operatorname{lcm}(\operatorname{LM}(f),
\operatorname{LM}(g))\).  Their \(S\)-polynomial is
\[
S(f,g)=\frac{M}{\operatorname{LT}(f)}f-
\frac{M}{\operatorname{LT}(g)}g,
\]
where division by a leading term includes division by its nonzero coefficient
in \(F\).  This combination cancels the leading terms.  We state Buchberger's criterion without proof: \(G\) is a Gr\"obner basis
exactly when every \(S(g_i,g_j)\) reduces to zero by \(G\).  This leads to
Buchberger's algorithm: start with generators; form all relevant
\(S\)-polynomials; reduce them; adjoin every nonzero remainder; and repeat
until all the remainders are zero.  Termination is a genuine theorem: each
new remainder strictly enlarges the leading-monomial ideal, and the ascending
chain condition for monomial ideals (Dickson's lemma) prevents this forever.

\begin{example}
In \(F[x,y]\) with lexicographic order \(x>y\), let
\(G=\{x-y,y^2-1\}\).  Their leading monomials are \(x,y^2\), and
\[
S(x-y,y^2-1)=y^2(x-y)-x(y^2-1)=x-y^3.
\]
Division first replaces \(x\) by \(y\), then replaces \(y^3\) by \(y\),
so the remainder is zero.  Buchberger's criterion shows that \(G\) is a
Gr\"obner basis.  Consequently \(x^2-1\) belongs to \((x-y,y^2-1)\): its
reduction is \(y^2-1\), then \(0\).  This small calculation illustrates why
a specially organized generating set makes ideal membership effective.
\end{example}

\section*{Exercises}
\begin{exercise}[Basic] Divide \(x^3-1\) by \(x-1\) and factor the result over \(\mathbb Q\). \end{exercise}
\begin{exercise}[Core] Use the Euclidean algorithm in \(\mathbb Q[x]\) to
find a gcd of \(x^3-1\) and \(x^2-1\), and express it as a B\'ezout combination.
\end{exercise}
\begin{exercise}[Core] Determine whether \(x^3-2\) and \(x^4+1\) are irreducible over \(\mathbb Q\). \end{exercise}
\begin{exercise}[Core] Find the content and primitive part of each of
\(12x^3-18x^2+6x\) and \(15x^2+10x+5\).  Verify the content formula for
their product. \end{exercise}
\begin{exercise}[Core] Prove directly from the definition that a primitive
polynomial in \(\mathbb Z[x]\) cannot have all of its coefficients divisible
by the same prime.  Use reduction modulo \(2\) to show that
\(x^3+x+1\) is irreducible over \(\mathbb Q\).
\emph{Hint:} use Gauss's Lemma; a nontrivial factorization of this primitive
polynomial would reduce to one in \(\mathbb F_2[x]\), where the cubic has no
root. \end{exercise}
\begin{exercise}[Further] Use B\'ezout to show that coprime polynomials in \(F[x]\) have no common root in any extension field. \end{exercise}
\begin{exercise}[Further, Challenging] Let \(R\) be a UFD and let
\(f\in R[x]\) be primitive.  Show that if \(f\) divides \(g\) in
\(\operatorname{Frac}(R)[x]\) and \(g\in R[x]\), then there is a
primitive polynomial \(h\in R[x]\) and \(a\in R\) such that
\(g=ahf\) up to multiplication by a unit. \end{exercise}
